{{- /* Video YouTube tải lười: trang chỉ hiện ảnh thumbnail và nút phát. Iframe của YouTube (vài trăm KB JavaScript) chỉ được tải khi người đọc bấm phát. {{< youtube id="dQw4w9WgXcQ" title="Tiêu đề hiện dưới video" >}} {{< youtube dQw4w9WgXcQ >}} ← viết gọn, không chú thích {{< youtube id="..." start="90" >}} ← bắt đầu từ giây thứ 90 Thumbnail được Hugo tải về lúc build rồi tự phục vụ từ nguyenta.com, nên trước khi bấm phát, trình duyệt người đọc không gửi gì tới Google. Khi phát thì dùng youtube-nocookie.com (chế độ tăng cường quyền riêng tư). */ -}} {{- $id := .Get "id" | default (.Get 0) -}} {{- $title := .Get "title" | default "" -}} {{- $start := .Get "start" | default "" -}} {{- $class := .Get "class" | default "" -}} {{- if not $id -}} {{- errorf "youtube: thiếu id video trong bài %q" .Page.Path -}} {{- end -}} {{- /* Thử ảnh lớn trước; video cũ không có maxresdefault thì lùi về hqdefault. Build không có mạng thì bỏ qua thumbnail, vẫn hiện khung và nút phát. */ -}} {{- $thumb := "" -}} {{- range slice "maxresdefault" "hqdefault" -}} {{- if not $thumb -}} {{- $url := printf "https://i.ytimg.com/vi/%s/%s.jpg" $id . -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err -}} {{- warnf "youtube: không tải được thumbnail %s" $url -}} {{- else with .Value -}} {{- /* Đặt tên theo id video, gom vào /video/ cho gọn */ -}} {{- $thumb = resources.Copy (printf "video/%s.jpg" $id) . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $watch := printf "https://www.youtube.com/watch?v=%s" $id -}}
